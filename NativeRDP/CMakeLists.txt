cmake_minimum_required(VERSION 3.21)
project(FjarrRDP LANGUAGES C CXX OBJC)
if(NOT APPLE)
  message(FATAL_ERROR "FjarrRDP requires macOS")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WITH_CLIENT_SDL OFF CACHE BOOL "" FORCE)
set(WITH_CLIENT_MAC OFF CACHE BOOL "" FORCE)
set(WITH_CLIENT_COMMON ON CACHE BOOL "" FORCE)
set(WITH_CLIENT ON CACHE BOOL "" FORCE)
set(WITH_SERVER OFF CACHE BOOL "" FORCE)
set(WITH_SAMPLE OFF CACHE BOOL "" FORCE)
set(WITH_X11 OFF CACHE BOOL "" FORCE)
set(WITH_WAYLAND OFF CACHE BOOL "" FORCE)
add_subdirectory("${FREERDP_SOURCE_DIR}" freerdp)
add_library(FjarrRDP SHARED FCRDPView.m)
target_compile_options(FjarrRDP PRIVATE -fobjc-arc -fvisibility=hidden)
target_include_directories(FjarrRDP PRIVATE "${FREERDP_SOURCE_DIR}/include" "${FREERDP_SOURCE_DIR}/winpr/include"
  "${CMAKE_CURRENT_BINARY_DIR}/freerdp/include" "${CMAKE_CURRENT_BINARY_DIR}/freerdp/winpr/include")
target_link_libraries(FjarrRDP PRIVATE freerdp-client "-framework AppKit" "-framework Carbon")
set_target_properties(FjarrRDP PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON)
# Explicit test target; the release contains only libFjarrRDP.dylib.
add_executable(FjarrRDPProbe EXCLUDE_FROM_ALL Tests/Probe.m)
target_compile_options(FjarrRDPProbe PRIVATE -fobjc-arc)
target_include_directories(FjarrRDPProbe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
  "${FREERDP_SOURCE_DIR}/include" "${FREERDP_SOURCE_DIR}/winpr/include"
  "${CMAKE_CURRENT_BINARY_DIR}/freerdp/include" "${CMAKE_CURRENT_BINARY_DIR}/freerdp/winpr/include")
target_link_libraries(FjarrRDPProbe PRIVATE FjarrRDP "-framework AppKit")
